  name: 'ticket',
  variants: ['card'],
  styles: `.aparte-ticket-card { padding: var(--aparte-rich-card-padding) }
           .aparte-ticket-card .t-id { font-weight: 700 }`,
  icon: '<svg viewBox="0 0 24 24">…</svg>',
  normalize: (raw) => (raw?.id ? raw : null),
  render: (t) => [
    el('div', { class: 't-id', text: `Ticket ${t.id}` }),
    el('p', { text: t.subject }),
    el('small', { text: t.status }),

  skeleton: () => [],

  fallback: (t) => `Ticket ${t.id}: ${t.subject}`,
});

  webhookUrl: '…',
  setup: (config) => ticket.setup({ config }),
